# SatyaVoice — Lightning AI Studio environment (TEMPLATE).
#
# Inside the Studio:
#   cp deploy/lightning/env.template.txt deploy/lightning/.env
#   nano deploy/lightning/.env
#   sh ./deploy/lightning/start.sh
#
# NEVER commit deploy/lightning/.env (it is gitignored via deploy/.gitignore).

# ---- Environment mode -------------------------------------------------------
# production enforces the full env contract (Postgres + auth keys required).
VOICETRUST_ENV=production

# ---- CORS: your Vercel frontend origin (exact, no trailing slash) ----------
VOICETRUST_CORS_ORIGINS=https://sih-voice.vercel.app

# ---- PostgreSQL (Neon free tier; keep ?sslmode=require) --------------------
VOICETRUST_DATABASE_URL=postgresql://user:password@ep-xxx.aws.neon.tech/neondb?sslmode=require

# ---- Auth ("tenant:key", comma-separated) ----------------------------------
VOICETRUST_AUTH_API_KEYS=sih-demo:sk_REPLACE_WITH_YOUR_GENERATED_KEY

# ---- Session store -----------------------------------------------------------
# Simplest inside a Studio: in-process memory (single uvicorn worker anyway).
VOICETRUST_SESSION_STORE_BACKEND=memory
# If you `apt install redis-server` inside the Studio, switch to redis:
# VOICETRUST_SESSION_STORE_BACKEND=redis
# VOICETRUST_REDIS_URL=redis://localhost:6379/0

# ---- AI modes (mock/manual first; real later if credits/RAM allow) ---------
VOICETRUST_DETECTOR_MODE=mock
VOICETRUST_ASR_MODE=manual
# IMPORTANT: facebook/wav2vec2-xls-r-300m is a BASE model (feature extractor),
# NOT an audio classifier - real mode will fail to load it. Use the
# placeholder checkpoint or your own fine-tuned classifier checkpoint.
VOICETRUST_MODEL_ID=Hemgg/Deepfake-audio-detection
VOICETRUST_ASR_MODEL_SIZE=small
VOICETRUST_ASR_DEVICE=cpu
VOICETRUST_ASR_COMPUTE_TYPE=int8

# ---- Speaker vault / logging ------------------------------------------------
VOICETRUST_SPEAKER_VAULT_ENABLED=true
VOICETRUST_RATE_LIMIT_ENABLED=true
VOICETRUST_LOG_LEVEL=INFO
